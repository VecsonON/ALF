ALF_events.1001 = {
	type = character_event
	title = ALF_events.1001.t
	desc = ALF_events.1001.desc

	theme = witchcraft

	option = { 
		trigger = {
			custom_tooltip = {
				text = ALF_events.1001.a.tt
				ALF_mental_fortitude >= var:ALF_greatest_spell_mental_fortitude
			}
		}
		show_as_unavailable = {
			custom_tooltip = {
				text = ALF_events.1001.a.tt
				ALF_mental_fortitude < var:ALF_greatest_spell_mental_fortitude
			}
		}
		name = ALF_events.1001.a
		ALF_greatest_spell_increase_mental_dif = yes
		ALF_greatest_spell_increase_success_chance = yes
		
		ALF_willpower_change_effect = { COUNT = 100 }
	}

	option = {
		name = ALF_events.1001.b
		trigger = {
			custom_tooltip = {
				text = ALF_events.1001.b.tt
				OR = {
					learning >= 20
					any_relation = { has_trait = immortal }
				}
			}
		}
		show_as_unavailable = {
			custom_tooltip = {
				text = ALF_events.1001.b.tt
				learning <= 20
				NOT = { any_relation = { has_trait = immortal } }
				
			}
		}
		ALF_greatest_spell_increase_success_chance = yes
	}

	option = {
		name = ALF_events.1001.c
	}

	after = {
		trigger_event = { id = ALF_events.1002 days = { 10 25 } }
	}
}

ALF_events.1005 = {
	type = character_event
	title = ALF_events.1005.t
	desc = ALF_events.1005.desc

	theme = witchcraft

	option = { 
		name = ALF_events.1005.a
		random_list = {
			0 = {
				desc = ALF_events.1005.a.a.tt
				modifier = { add = var:ALF_greatest_spell_success_chance }
				add_trait = immortal
			}
			0 = {
				desc = ALF_events.1005.a.b.tt
				modifier = { add = var:ALF_greatest_spell_success_chance }
				modifier = { add = var:ALF_greatest_spell_success_chance }
				remove_list_variable = {
					name = unavailable_unique_decisions
					target = flag:ALF_greatest_spell_objective_decision
				}
			}
			100 = {
				desc = ALF_events.1005.a.c.tt
				modifier = { add = { value = var:ALF_greatest_spell_success_chance multiply = -1 } }
				modifier = { add = { value = var:ALF_greatest_spell_success_chance multiply = -1 } }
				modifier = { add = { value = var:ALF_greatest_spell_success_chance multiply = -1 } }
				custom_tooltip =  ALF_events.1005.a.c.effect
				set_global_variable = {
					name = the_end_of_the_world
					value = yes
				}
			}
		}
	}

	option = {
		name = ALF_events.1005.b
		remove_list_variable = {
			name = unavailable_unique_decisions
			target = flag:ALF_greatest_spell_objective_decision
		}
	}

	after = {
		if = { limit = { has_variable = ALF_greatest_spell_mental_fortitude } remove_variable = ALF_greatest_spell_mental_fortitude }
		if = { limit = { has_variable = ALF_greatest_spell_success_chance } remove_variable = ALF_greatest_spell_success_chance }
		
	}
}
