king_of_fire_perk = {
	lifestyle = willmagic_lifestyle
	tree = elementis
	position = { 0 6.8 }
	icon = ALF_willmagic

	parent = king_of_elements_perk

	effect = {
		custom_description_no_bullet = { text = repeatable_perk_effect }
		custom_description_no_bullet = { text = king_of_fire_perk_effect }
		ALF_set_variable_or_change = { VAR = ALF_king_of_fire_perk_count_var VAL = 1 }
		hidden_effect = { 
			remove_character_modifier = king_of_fire_modifier
			remove_perk = king_of_fire_perk 
		}
		add_character_modifier = king_of_fire_modifier
	}
}
