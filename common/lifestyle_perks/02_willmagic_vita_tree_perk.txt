long_life_perk = {
	lifestyle = willmagic_lifestyle
	tree = vita
	position = { 1 6.8 }
	icon = ALF_willmagic

	parent = improve_flesh_perk

	can_be_picked = {
		willmagic_lifestyle_perk_points >= long_life_additional_cost
	}

	effect = {
		custom_description_no_bullet = { text = repeatable_perk_effect }
		custom_description_no_bullet = { text = long_life_perk_effect }
		custom_description_no_bullet = { 
			text = long_life_perk_cost_effect 
			value = long_life_additional_cost
			save_scope_value_as = { name = calc2 value = { value = long_life_additional_cost subtract = 1 multiply = -1 } }
			hidden_effect = { add_willmagic_lifestyle_perk_points = scope:calc2 }
		}
		ALF_set_variable_or_change = { VAR = ALF_long_life_perk_count_var VAL = 1 }
		hidden_effect = {
			remove_character_modifier = long_life_modifier
			remove_perk = long_life_perk
		}
		add_character_modifier = long_life_modifier
	}
}
