ALF_teach_willmagic_interaction = {
	category = interaction_category_friendly
	icon = ALF_willmagic
	desc = ALF_teach_willmagic_interaction_desc

	interface_priority = 1
	
	common_interaction = yes

	cooldown = { years = 5 }
	is_shown = {
		scope:actor = { has_trait = ALF_mage }
		scope:recipient = { NOT = { has_trait = ALF_mage } }
		scope:recipient = { age >= 13 }
		NOT = { scope:actor = scope:recipient }
	}

	on_accept = {
		scope:recipient = {
			random = {
				chance = {
					value = ALF_mental_fortitude
					multiply = 10
					add = sum_of_all_skills_value
				}
				add_trait = ALF_mage
			}
			add_stress = 75
		}
		scope:actor = {
			ALF_use_willpower_effect = { COUNT = 50 }
		}
	}

	ai_accept = {
		base = 0

		modifier = { # Greedy for power
			add = 50
			scope:recipient = {
				has_trait = ambitious
				NOT = { has_trait = zealous }
			}
			desc = ambition_tt
		}
		opinion_modifier = { # Compare Opinion modifier.
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.6
		}
		modifier = { # Say no to rivals
			add = -25
			scope:recipient = {
				has_relation_rival = scope:actor
			}
			desc = "ACTOR_RIVAL_TO_ME_REASON"
		}
		modifier = { # Say no to nemesis
			add = -50
			scope:recipient = {
				has_relation_nemesis = scope:actor
			}
			desc = "ACTOR_NEMESIS_TO_ME_REASON"
		}
		modifier = {
			add = 1000
			scope:recipient = {
				is_heir_of = scope:actor
			}
			desc = "is_recipients_player_heir"
		}
	}
}
