ALF_open_gui_character_interaction = {
	category = interaction_category_diplomacy
	icon = ALF_willmagic
	desc = ALF_open_gui_interaction_desc

	common_interaction = yes

	is_shown = {
		scope:actor = { ALF_has_access_to_willmagic_trigger = yes }
	}

	on_accept = {
		scope:actor = {
			ALF_open_willmagic_gui_effect = { TARGET = scope:recipient TYPE = character }
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
