ALF_AI_close_wound_interaction = {
	category = interaction_category_friendly

	is_shown = { 
		scope:actor = {
			has_perk = close_wound_perk
			is_ai = yes
		}
		scope:recipient = {
			has_trait = wounded
		}
	}

	on_accept = {
		scope:actor = { 
			ALF_ai_spell_cast_effect = { SPELL = close_wound }
		}
	}

	auto_accept = yes
	
	ai_potential = {
		is_imprisoned = no 
		has_trait = ALF_mage
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = scripted_relations
		max = 5
	}
	ai_frequency = 6
	ai_will_do = {
		base = 50

		modifier = { # AI will more likely to heal themselves than others
			scope:recipient = scope:actor
			add = 250
		}

		modifier = { # AI won't heal rivals or nemesis.
			scope:recipient = {
				OR = {
					has_relation_rival = scope:actor
					has_relation_nemesis = scope:actor 
				}
			}
			add = -1000
		}

		opinion_modifier = { # AI will more likely heal characters with better opinion
			trigger = {
				NOT = { scope:recipient = scope:actor }
			}
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			min = 0
		}
	}
}
