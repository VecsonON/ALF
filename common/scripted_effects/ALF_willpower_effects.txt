ALF_set_mental_fortitude_from_parents_effect = {
	random_list = {
		1 = {
			trigger = { exists = mother mother ?= { is_alive = yes } }
			set_variable = { name = ALF_mental_fortitude_genetic_var value = mother.ALF_mental_fortitude_genetic }
		}
		1 = {
			trigger = { exists = real_father real_father ?= { is_alive = yes } }
			set_variable = { name = ALF_mental_fortitude_genetic_var value = real_father.ALF_mental_fortitude_genetic }
		}
	}
	random_list = {
		1 = { change_variable = { name = ALF_mental_fortitude_genetic_var add = -2 } }
		3 = { change_variable = { name = ALF_mental_fortitude_genetic_var add = -1 } }
		5 = { change_variable = { name = ALF_mental_fortitude_genetic_var add = 0 } }
		3 = { change_variable = { name = ALF_mental_fortitude_genetic_var add = 1 } }
		1 = { change_variable = { name = ALF_mental_fortitude_genetic_var add = 2 } }
	}
}
ALF_change_mental_fortitude_effect = {
	if = { 
		limit = { $COUNT$ > 0 } 
		custom_description = {
			text = ALF_add_mental_fortitude_effect
			value = $COUNT$
		}
	}
	else = {
		custom_description = {
			text = ALF_subtract_mental_fortitude_effect
			value = $COUNT$
		}
	}
	ALF_set_variable_or_change = { VAR = ALF_mental_fortitude_var VAL = $COUNT$ }
}
ALF_willpower_change_effect = {
	if = { 
		limit = { $COUNT$ > 0 } 
		custom_description = {
			text = ALF_willpower_cost_effect
			value = $COUNT$
		}
	}
	else = {
		custom_description = {
			text = ALF_willpower_regain_effect
			value = $COUNT$
		}
	}
	hidden_effect = {
		if = { limit = { NOT = { $COUNT$ = 0 } } 
			add_character_flag = ALF_willpower_change_flag
			save_scope_value_as = { name = saved_stress  value = stress }
			
			add_stress = 100
			
			while = {
				limit = { stress = scope:saved_stress }
				add_character_modifier = ALF_will_power_1_modifier
				force_character_skill_recalculation = yes
				add_stress = 100
			}
			add_stress = -600
			while = {
				limit = { stress > 0 }
				add_character_modifier = ALF_will_power_2_modifier
				force_character_skill_recalculation = yes
				add_stress = -600
			}      
			add_stress = 100
			while = {
				limit = { NOT = { stress = 100 } }
				if = { limit = { stress > 100 } add_character_modifier = ALF_will_power_3_modifier }
				else = { add_character_modifier = ALF_will_power_35_modifier }

				force_character_skill_recalculation = yes
				add_stress = -600
				add_stress = 100
			}
			add_stress = -600
			add_stress = 100
			add_stress = -50
			while = {
				limit = { NOT = { stress = 50 } }
				if = { limit = { stress < 50 } add_character_modifier = ALF_will_power_4_modifier }
				else = { add_character_modifier = ALF_will_power_45_modifier }
				force_character_skill_recalculation = yes

				add_stress = -600
				add_stress = 100
				add_stress = -50
			}
			add_stress = -600
			add_stress = scope:saved_stress
			remove_character_flag = ALF_willpower_change_flag
			add_stress = $COUNT$
			remove_all_character_modifier_instances = ALF_will_power_1_modifier
			remove_all_character_modifier_instances = ALF_will_power_2_modifier
			remove_all_character_modifier_instances = ALF_will_power_3_modifier
			remove_all_character_modifier_instances = ALF_will_power_35_modifier
			remove_all_character_modifier_instances = ALF_will_power_4_modifier
			remove_all_character_modifier_instances = ALF_will_power_45_modifier
			force_character_skill_recalculation = yes
		}
	}
}

ALF_use_willpower_effect = {
	save_scope_value_as = { 
		name = true_value 
		value = { 
			value = $COUNT$ 
			subtract = ALF_willpower_cost_flat 
			multiply = ALF_willpower_owned_cost_mult 
		} 
	}
	ALF_willpower_change_effect = { COUNT = scope:true_value }
}

ALF_use_willpower_with_enemy_effect = {
	if = { 
		limit = { ROOT = $TARGET$ } 
		ALF_use_willpower_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = { 
			name = true_value 
			value = { 
				value = $COUNT$ 
				subtract = ALF_willpower_cost_flat 
				multiply = ALF_willpower_owned_cost_mult 
				multiply = $TARGET$.ALF_willpower_hostile_cost_mult 
			} 
		}
		ALF_willpower_change_effect = { COUNT = scope:true_value }
	}
}
