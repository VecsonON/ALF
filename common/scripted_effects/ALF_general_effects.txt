ALF_set_variable_or_change = {
	if = { limit = { has_variable = $VAR$ } change_variable = { name = $VAR$ add = $VAL$ } }
	else = { set_variable = { name = $VAR$ value = $VAL$ } }
}

ALF_assign_perk = {
	if = {
		limit = { NOT = { has_perk = $PERK$ } }
		add_perk = $PERK$
	}
}
