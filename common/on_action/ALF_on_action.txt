on_birth_child = { on_actions = { ALF_on_birth_child } }
on_game_start_after_lobby = { on_actions = { ALF_on_game_start_after_lobby } }
yearly_global_pulse = { on_actions = { ALF_yearly_global_pulse } }

ALF_on_birth_child = {
	effect = {
		ALF_set_mental_fortitude_from_parents_effect = yes
	}
}
ALF_on_game_start_after_lobby = {
	effect = {
		every_living_character = {
			limit = {  
				NOR = {
					mother ?= { is_alive = yes }
					real_father ?= { is_alive = yes }
				}
			}
			save_scope_value_as = { name = random_mental_fortitude value = { value = { 0 10 } floor = yes } }
			set_variable = { name = ALF_mental_fortitude_genetic_var value = scope:random_mental_fortitude }
			every_child = {
				limit = { is_alive = yes }
				ALF_set_mental_fortitude_from_parents_effect = yes
				ALF_change_mental_fortitude_effect = { COUNT = ALF_mental_fortitude_from_living }
			}
			ALF_change_mental_fortitude_effect = { COUNT = ALF_mental_fortitude_from_living }
		}
		every_player = {
			trigger_event = ALF_events.0001
		}
	}
}
ALF_yearly_global_pulse = {
	effect = {
		every_living_character = {
			if = {
				limit = {
					save_temporary_scope_value_as = {
						name = biyearly_countdown
						value = { value = current_year modulo = 2 }
					}
					scope:biyearly_countdown = 0
				}
				if = { limit = { has_perk = regenerate_will_perk } 
					send_interface_message = {
						title = regenerate_will_perk_name
						ALF_change_mental_fortitude_effect = { COUNT = 2 }
					}
				}
				else = {
					send_interface_message = {
						title = ALF_yearly_global_pulse_mental_fortitude
						ALF_change_mental_fortitude_effect = { COUNT = 1 }
					}
				}
			}
			if = {
				limit = { 
					is_ai = yes 
					has_trait = ALF_mage
					NOT = { has_focus = willmagic_focus } 
				}
				random = {
					chance = 1 # Sometimes AI will switch to willmagic
					set_focus = willmagic_focus
				}
			}
		}
	}
}

ALF_stress_pulse = {
	effect = {
		random = {
			chance = ALF_stress_break_mental_fortitude_increase_chance
			send_interface_message = {
				title = train_will_perk_name
				ALF_change_mental_fortitude_effect = { COUNT = 1 }
			}
		}
	}
}
