on_birth_child = { on_actions = { ALF_on_birth_child } }
on_game_start_after_lobby = { on_actions = { ALF_on_game_start_after_lobby } }
yearly_global_pulse = { on_actions = { ALF_yearly_global_pulse } }

ALF_on_birth_child = {
	effect = {
		ALF_set_mental_fortitude_from_parents_effect = yes
	}
}
ALF_on_game_start_after_lobby = {
	effect = {
		every_living_character = {
			limit = {  
				NOR = {
					mother ?= { is_alive = yes }
					real_father ?= { is_alive = yes }
				}
			}
			save_scope_value_as = { name = random_mental_fortitude value = { value = { 0 10 } floor = yes } }
			set_variable = { name = ALF_mental_fortitude_genetic_var value = scope:random_mental_fortitude }
			every_child = {
				limit = { is_alive = yes }
				ALF_set_mental_fortitude_from_parents_effect = yes
				ALF_change_mental_fortitude_effect = { COUNT = ALF_mental_fortitude_from_living }
			}
			ALF_change_mental_fortitude_effect = { COUNT = ALF_mental_fortitude_from_living }
		}
		every_player = {
			trigger_event = ALF_events.0001
		}

		random_culture_global = {
			save_scope_as = culture
		}
		random_religion_global = {
			random_faith = {
				save_scope_as = faith
			}
		}
		random_province = {
			save_scope_as = location
		}

		create_character = {
			age = { 500 1000 }
			name = ALF_unnamed
			dynasty = none
			gender_female_chance = 0
			random_traits = no
			location = scope:location
			faith = scope:faith
			culture = scope:culture
			trait = education_learning_5

			trait = patient
			trait = calm
			trait = cynical
			trait = lifestyle_scholar
			trait = whole_of_body
			trait = architect
			trait = immortal
			trait = ALF_mage
			trait = intellect_good_3
			
			diplomacy = 10
			martial = 10
			stewardship = 15
			intrigue = 10
			learning = 20
			prowess = 20

			after_creation = {
				set_immortal_age = 22
				set_variable = { name = ALF_mental_fortitude_genetic_var value = 11 }
				ALF_change_mental_fortitude_effect = { COUNT = 125 }

				ALF_assign_voluntas_perks_effect = { LEVEL = 7 }
				ALF_assign_animalium_perks_effect = { LEVEL = 7 }
				ALF_assign_vita_perks_effect = { LEVEL = 7 }
				ALF_assign_elementis_perks_effect = { LEVEL = 7 TYPE = fire }
				ALF_assign_elementis_perks_effect = { LEVEL = 7 TYPE = wind }
				ALF_assign_elementis_perks_effect = { LEVEL = 7 TYPE = water }
				ALF_assign_elementis_perks_effect = { LEVEL = 7 TYPE = rock }
				ALF_assign_fortuna_perks_effect = { LEVEL = 7 }
				ALF_assign_aspectum_perks_effect = { LEVEL = 7 }
			}
		}
	}
}
ALF_yearly_global_pulse = {
	effect = {
		every_living_character = {
			if = {
				limit = {
					save_temporary_scope_value_as = {
						name = biyearly_countdown
						value = { value = current_year modulo = 2 }
					}
					scope:biyearly_countdown = 0
				}
				if = { limit = { has_perk = regenerate_will_perk } 
					send_interface_message = {
						title = regenerate_will_perk_name
						ALF_change_mental_fortitude_effect = { COUNT = 2 }
					}
				}
				else = {
					send_interface_message = {
						title = ALF_yearly_global_pulse_mental_fortitude
						ALF_change_mental_fortitude_effect = { COUNT = 1 }
					}
				}
			}
			if = {
				limit = { 
					is_ai = yes 
					has_trait = ALF_mage
					NOT = { has_focus = willmagic_focus } 
				}
				random = {
					chance = 1 # Sometimes AI will switch to willmagic
					set_focus = willmagic_focus
				}
			}
		}
	}
}

ALF_stress_pulse = {
	effect = {
		random = {
			chance = ALF_stress_break_mental_fortitude_increase_chance
			send_interface_message = {
				title = train_will_perk_name
				ALF_change_mental_fortitude_effect = { COUNT = 1 }
			}
		}
	}
}
