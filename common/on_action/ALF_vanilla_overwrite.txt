# Triggered when stress level goes down
# Root is the character
on_stress_level_reduced = {
	trigger = {
		NOT = { has_character_flag = recently_hit_stress_threshold }
	}
}
#Going to a new stress level will fire this.
on_stress_level_1 = {
	on_actions = {
		ALF_stress_pulse
	}
}
on_stress_level_2 = {
	on_actions = {
		ALF_stress_pulse
	}
}
on_stress_level_3 = {
	on_actions = {
		ALF_stress_pulse
	}
}
on_stress_level_4 = {
	on_actions = {
		ALF_stress_pulse
	}
}

stress_threshold_level_1_event = {
	random_events = {
		500 = ALF_events.1
	}
}
stress_threshold_level_2_event = {
	random_events = {
		800 = ALF_events.1
	}
}
stress_threshold_level_3_event = {
	random_events = {
		1000 = ALF_events.1
	}
}
stress_threshold_level_4_event = {
	random_events = {
		2000 = ALF_events.1
	}
}
