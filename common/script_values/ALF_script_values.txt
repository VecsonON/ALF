# Requires scope:count and optional scope:target
ALF_willpower_cost = {
	value = scope:count
	multiply = ALF_willpower_cost_mult
	subtract = ALF_willpower_cost_flat
	min = 0
	divide = 2
	floor = yes
}

ALF_willpower_cost_mult = {
	value = 1

	add = ALF_willpower_cost_caster_mult
	
	if = { 
		limit = { exists = scope:target } 
		scope:target = { 
			add = ALF_willpower_cost_target_mult
		}
	}
	min = 0
}

ALF_willpower_cost_caster_mult = {
	if = { limit = { has_perk = command_will_perk } add = -0.05 }
	if = { limit = { has_perk = extend_will_perk } add = -0.05 }
	if = { limit = { has_perk = attack_will_perk } add = -0.1 }

	add = {
		value = enforce_will_perk_count_tt
		multiply = -0.05
	}
}

ALF_willpower_cost_target_mult = {
	if = { limit = { has_perk = command_will_perk } add = 0.05 }
	if = { limit = { has_perk = extend_will_perk } add = 0.05 }
	if = { limit = { has_perk = defend_will_perk } add = 0.1 }

	add = {
		value = counter_will_perk_count_tt
		multiply = 0.05
	}
}
