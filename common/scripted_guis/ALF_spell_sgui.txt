inspect_will_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		trigger_if = { 
			limit = { ALF_general_spell_trigger = yes } 
			var:spell_target ?= { NOT = { has_perk = hide_will_perk } } 
			trigger_if = { limit = { NOT = { var:spell_target = THIS } } has_perk = inspect_will_perk }
		}
	}
	effect = {
		if = { 
			limit = { ALF_general_spell_trigger = yes } 
			ALF_inspect_will_effect = yes
		}
	}
}
steal_will_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		has_perk = steal_will_perk
		trigger_if = { 
			limit = { ALF_general_spell_trigger = yes } 
			var:spell_target ?= { is_imprisoned_by = PREV } 
			custom_description = { 
				text = ALF_target_has_higher_willpower_trigger
				object = var:spell_target
				var:spell_target.ALF_mental_fortitude >= ALF_mental_fortitude
			}
		}
	}
	effect = {
		if = { 
			limit = { ALF_general_spell_trigger = yes } 
			ALF_steal_will_effect = yes
		}
	}
}
