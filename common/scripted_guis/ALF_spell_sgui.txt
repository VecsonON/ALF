inspect_will_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		trigger_if = { 
			limit = { ALF_general_spell_trigger = yes }  
			trigger_if = { 
				limit = { NOT = { var:spell_target = THIS } }
				has_perk = inspect_will_perk 
				var:spell_target ?= { NOT = { has_perk = hide_will_perk } }
			}
		}
	}
	effect = {
		if = { 
			limit = { ALF_general_spell_trigger = yes } 
			ALF_inspect_will_effect = yes
		}
	}
}
steal_will_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		has_perk = steal_will_perk
		trigger_if = { 
			limit = { ALF_general_spell_trigger = yes } 
			var:spell_target ?= { is_imprisoned_by = PREV } 
			custom_description = { 
				text = ALF_target_has_higher_willpower_trigger
				object = var:spell_target
				var:spell_target.ALF_mental_fortitude >= ALF_mental_fortitude
			}
		}
	}
	effect = {
		if = { 
			limit = { ALF_general_spell_trigger = yes } 
			ALF_steal_will_effect = yes
		}
	}
}

bless_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		trigger_if = { 
			limit = { var:spell_type ?= flag:character } 
			var:spell_target = { NOT = { has_character_modifier = bless_character_modifier } }
			has_perk = bless_character_perk
		}
		trigger_if = { 
			limit = { var:spell_type ?= flag:artifact } 
			custom_tooltip = { 
				text = spell_bless_tt 
				var:spell_target = { NOT = { has_artifact_modifier = bless_artifact_modifier } } 
			}
			has_perk = bless_artifact_perk
			
		}
		trigger_if = { 
			limit = { var:spell_type ?= flag:title } 
			var:spell_target = { tier = 2 }
			trigger_if = { 
				limit = { var:spell_target = { tier = 2 } } 
				var:spell_target = { NOT = { has_county_modifier = bless_land_modifier } }
			}
			has_perk = bless_land_perk
		}
	}
	effect = {
		if = { 
			limit = { 
				ALF_general_spell_trigger = yes 
				trigger_if = { 
					limit = { var:spell_type ?= flag:title } 
					var:spell_target = { tier = 2 }
				}
			} 
			ALF_bless_effect = yes
		}
	}
}
