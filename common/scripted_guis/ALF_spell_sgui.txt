inspect_will_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		trigger_if = { 
			limit = { var:spell_type ?= flag:character } 
			var:spell_target ?= { NOT = { has_perk = hide_will_perk } } 
			trigger_if = { limit = { NOT = { var:spell_target = THIS } } has_perk = inspect_will_perk }
		}
	}
	effect = {
		if = { 
			limit = { var:spell_type ?= flag:character } 
			ALF_inspect_will_effect = yes
		}
	}
}
steal_will_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		has_perk = steal_will_perk
		trigger_if = { 
			limit = { var:spell_type ?= flag:character } 
			var:spell_target ?= { is_imprisoned_by = PREV } 
			custom_description = { 
				text = ALF_target_has_higher_willpower_trigger
				object = var:spell_target
				var:spell_target.ALF_mental_fortitude >= ALF_mental_fortitude
			}
		}
	}
	effect = {
		if = { 
			limit = { var:spell_type ?= flag:character } 
			ALF_steal_will_effect = yes
		}
	}
}
influence_animals_sgui = {
	is_valid = { 
		ALF_general_spell_trigger = yes
		has_perk = influence_animals_perk
		trigger_if = { 
			limit = { var:spell_type ?= flag:character } 
			ALF_target_is_root = yes
		}
		custom_tooltip = {
			text = spell_influence_animals_tt
			NOT = {
				AND = {
					has_character_modifier = cat_story_modifier
					has_character_modifier = dog_story_modifier
					has_character_modifier = horse_story_modifier
				}
			}
		}
	}
	effect = {
		if = { 
			limit = { var:spell_type ?= flag:character } 
			ALF_influence_animals_effect = yes
		}
	}
}
