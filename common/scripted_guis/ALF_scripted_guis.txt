	effect = {
		remove_variable = willmagic_gui

ALF_spell_target_character = { 
	is_valid = { 
		OR = {
			scope:spell.var:type ?= flag:character 
			scope:spell.var:type ?= flag:any 
		}
	} 
}

ALF_spell_target_is_you = {
	is_valid = {
		trigger_if = { limit = { var:spell_type ?= flag:character } var:spell_target = ROOT }
		trigger_else_if = { limit = { var:spell_type ?= flag:artifact } var:spell_target.artifact_owner = ROOT }
		trigger_else = { var:spell_target.holder = ROOT }
	}
}

ALF_clearup = {
	effect = {
		clear_variable_list = WillMagicPaths
		every_in_list = {
			variable = WillMagicSpells
			end_story = yes
		}
		clear_variable_list = WillMagicSpells
		remove_variable = spell
	}
}
ALF_filter_set = { 
	saved_scopes = { filter }
	effect = { 
		set_variable = { name = spell_filter value = scope:filter } 
		if = { limit = { var:spell_filter ?= flag:any } }
	} 
}
ALF_filter_check = {
	saved_scopes = { spell_filter }
	is_shown = {
		trigger_if = { limit = { var:spell_filter ?= flag:any } }
		trigger_else_if = {
			limit = { 
				OR = {
					var:spell_filter ?= flag:character 
					var:spell_filter ?= flag:artifact 
					var:spell_filter ?= flag:title 
				}
			}
			OR = {
				scope:spell.var:type ?= var:spell_filter
				scope:spell.var:type ?= flag:any
			}
		}
		trigger_else = {
			scope:spell.var:path ?= var:spell_filter
		}
	} 
}
ALF_select_spell = {

	}
