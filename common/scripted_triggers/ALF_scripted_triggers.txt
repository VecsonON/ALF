ALF_has_access_to_willmagic_trigger = {
	OR = {
		has_trait_with_flag = ALF_has_access_to_willmagic
		# has_character_flag = ALF_has_access_to_willmagic
	}
}
ALF_general_spell_trigger = {
	custom_tooltip = { 
		text = ALF_cast_spell_sgui_type
		OR = {
			scope:spell.var:type ?= var:spell_type
			scope:spell.var:type ?= flag:any
		}
	}
	custom_tooltip = { 
		text = ALF_cast_spell_sgui_stress
		stress_level <= 2
	}
}
ALF_target_is_root = {
	custom_tooltip = { 
		text = ALF_target_is_root
		var:spell_target = ROOT
	}
}

ALF_master_mage_trigger = {
	OR = {
		has_perk = conquer_will_perk # Voluntas
		has_perk = supernatural_animals_perk # Animalium
		has_perk = improve_flesh_perk # Vita
		has_perk = king_of_elements_perk # Elementis
		has_perk = tragic_day_perk # Fortuna
		has_perk = scry_for_education_perk # Aspectum
	}
}
